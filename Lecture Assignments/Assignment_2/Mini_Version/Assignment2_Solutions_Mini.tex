\section*{Problem 1: Zero-Inflated Normal}

\textbf{(a) PMF/PDF and CDF:}

PDF (mixture distribution):
\[
f_X(x) = \begin{cases}
\alpha & \text{at } x=0 \text{ (point mass)} \\
(1-\alpha) \cdot \frac{1}{\sigma\sqrt{2\pi}} e^{-(x-\mu)^2/2\sigma^2} & x \neq 0
\end{cases}
\]

CDF:
\[
F_X(x) = \begin{cases}
(1-\alpha)\Phi\left(\frac{x-\mu}{\sigma}\right) & x < 0 \\[6pt]
\alpha + (1-\alpha)\Phi\left(\frac{x-\mu}{\sigma}\right) & x \geq 0
\end{cases}
\]

where \(\Phi\) is standard normal CDF.

\textbf{(b) Mean and Variance:}

Expected value:
\begin{align*}
\mathbb{E}[X] &= 0 \cdot \alpha + \mu \cdot (1-\alpha) \\
&= \boxed{(1-\alpha)\mu}
\end{align*}

For variance, compute \(\mathbb{E}[X^2]\):
\begin{align*}
\mathbb{E}[X^2] &= (1-\alpha)(\sigma^2 + \mu^2)
\end{align*}

Then:
\begin{align*}
\text{Var}(X) &= \mathbb{E}[X^2] - (\mathbb{E}[X])^2 \\
&= (1-\alpha)(\sigma^2 + \mu^2) - (1-\alpha)^2\mu^2 \\
&= (1-\alpha)\sigma^2 + (1-\alpha)\alpha\mu^2 \\
&= \boxed{(1-\alpha)(\sigma^2 + \alpha\mu^2)}
\end{align*}

\textbf{(c) Numerical:} \(\alpha=0.3, \mu=2, \sigma=1\)

\begin{align*}
\mathbb{P}(X \leq 0) &= \mathbb{P}(X = 0) + \mathbb{P}(X < 0) \\
&= \alpha + (1-\alpha)\Phi\left(\frac{-\mu}{\sigma}\right) \\
&= 0.3 + 0.7 \cdot \Phi(-2) \\
&= 0.3 + 0.7(0.0228) \\
&= \boxed{0.316}
\end{align*}

\section*{Problem 2: Binomial \& Normal Approximation}

Given: \(X \sim \text{Bin}(10, 0.3)\)

\textbf{(a) Exact:}

\begin{align*}
\mathbb{P}(X \geq 6) &= \sum_{k=6}^{10} \binom{10}{k}(0.3)^k(0.7)^{10-k}
\end{align*}

Calculations:
\begin{align*}
P(X=6) &= 210(0.3)^6(0.7)^4 = 0.0368 \\
P(X=7) &= 120(0.3)^7(0.7)^3 = 0.0090 \\
P(X=8) &= 45(0.3)^8(0.7)^2 = 0.0014 \\
P(X=9) &= 10(0.3)^9(0.7)^1 = 0.0001 \\
P(X=10) &= 1(0.3)^{10} = 0.000006
\end{align*}

Sum: \(\boxed{\mathbb{P}(X \geq 6) = 0.0473}\)

\textbf{(b) Normal Approximation:}

Parameters: \(\mu = np = 3\), \(\sigma^2 = np(1-p) = 2.1\), \(\sigma = 1.449\)

With continuity correction:
\begin{align*}
\mathbb{P}(X \geq 6) &\approx \mathbb{P}(Y > 5.5) \\
&= \mathbb{P}\left(Z > \frac{5.5-3}{1.449}\right) \\

&= 1 - 0.9582 = \boxed{0.0418}
\end{align*}

Error: \(|0.0473 - 0.0418| = 0.0055\) (11.6\%)

\section*{Problem 3: Poisson Thinning}

\textbf{(a) Thinning Property:}

Given \(N \sim \text{Poisson}(\lambda)\), each event tagged "A" with prob. \(q\).

\begin{align*}
\mathbb{P}(K=k) &= \sum_{n=k}^{\infty} \binom{n}{k}q^k(1-q)^{n-k} \frac{\lambda^n e^{-\lambda}}{n!} \\
&= \frac{q^k e^{-\lambda}}{k!} \sum_{n=k}^{\infty} \frac{\lambda^n(1-q)^{n-k}}{(n-k)!}
\end{align*}

Let \(m=n-k\):
\begin{align*}
&= \frac{(\lambda q)^k e^{-\lambda}}{k!} \sum_{m=0}^{\infty} \frac{[\lambda(1-q)]^m}{m!} \\
&= \frac{(\lambda q)^k e^{-\lambda}}{k!} \cdot e^{\lambda(1-q)} \\
&= \frac{(\lambda q)^k}{k!} e^{-\lambda q}
\end{align*}

Thus: \(\boxed{K \sim \text{Poisson}(\lambda q)}\)

\textbf{(b) Conditional Distribution:}

Given \(N=n\), each of \(n\) events independently tagged "A" with prob. \(q\):

\(\boxed{K \mid N=n \sim \text{Bin}(n,q)}\)

\textbf{(c) Numerical:} \(\lambda=12, q=0.2\), so \(\lambda q = 2.4\)

\begin{align*}
\mathbb{P}(K=0) &= e^{-2.4} = \boxed{0.0907} \\
\mathbb{P}(K \leq 2) &= e^{-2.4}\left(1 + 2.4 + \frac{2.4^2}{2}\right) \\
&= 0.0907(1 + 2.4 + 2.88) \\
&= \boxed{0.570}
\end{align*}

\section*{Problem 5: Binning with Outlier}

Data: 199 points \(\sim N(0,1)\), outlier \(x_{200}=8\)

\(\sum_{i=1}^{199} x_i = 0\), \(\sum_{i=1}^{199} x_i^2 = 199\)

\textbf{(a) Statistics with outlier:}

\begin{align*}
\bar{x} &= \frac{0 + 8}{200} = 0.04 \\
\sum x_i^2 &= 199 + 64 = 263
\end{align*}

\begin{align*}
s^2 &= \frac{263}{200} - (0.04)^2 = 1.315 - 0.0016 = 1.3134 \\
s &= 1.146
\end{align*}

\textbf{(b) Bin widths:}

Sturges: \(K = \lceil \log_2(200) \rceil + 1 = \lceil 7.64 \rceil + 1 = 9\)

Scott:
\begin{align*}
h_S &= 3.5 \cdot s \cdot n^{-1/3} \\
&= 3.5(1.146)(200)^{-1/3} \\
&= 3.5(1.146)(0.1716) = 0.688
\end{align*}

FD (using \(\text{IQR} \approx 1.349\) for standard normal):
\begin{align*}
h_{FD} &= 2 \cdot \text{IQR} \cdot n^{-1/3} \\
&= 2(1.349)(0.1716) = 0.463
\end{align*}

\textbf{(c) Bins on \([-4,8]\) (range=12):}

\begin{align*}
\text{Sturges:} &\quad 12/1.33 = 9 \text{ bins} \\
\text{Scott:} &\quad 12/0.688 \approx 17 \text{ bins} \\
\text{FD:} &\quad 12/0.463 \approx 26 \text{ bins}
\end{align*}

FD most robust (IQR unaffected by single outlier).

\section*{Problem 6: Logistic Regression (2×2)}

\textbf{(a) Coefficients:}

\section*{Problem 8: Impurity Measures}

\textbf{(a) Prove \(H(p) \geq G(p)\):}

\(H(p) = -p\log_2 p - (1-p)\log_2(1-p)\)

\(G(p) = 2p(1-p)\)

At \(p=0.5\): \(H(0.5) = 1\), \(G(0.5) = 0.5\)

By calculus: \(H(p) - G(p) \geq 0\) for all \(p \in [0,1]\)

\textbf{(b) Prove \(G(p) \geq E(p)\):}

\(E(p) = \min\{p, 1-p\}\)

For \(p \leq 0.5\): \(G(p) = 2p(1-p) \geq p = E(p)\)

since \(2(1-p) \geq 1\) when \(p \leq 0.5\).

\(G(S) = 2(0.5)(0.5) = 0.5\)

\textbf{Attribute A} (\(v_1\): 4/1, \(v_2\): 1/2, \(v_3\): 1/3):

\begin{align*}
H(S|A) &= \frac{5}{12}H(0.8) + \frac{3}{12}H(0.33) + \frac{4}{12}H(0.25) \\

\end{align*}

\section*{Problem 10: Simple Linear Regression}

Data: \((0,1.0), (1,2.9), (2,5.1), (3,5.9), (4,9.2)\)

\textbf{(a) OLS Estimates:}

\(\bar{x} = 2\), \(\bar{y} = 4.82\)

\(S_{xx} = 10\), \(S_{xy} = 19.4\), \(S_{yy} = 38.708\)

\begin{align*}
\hat{\beta}_1 &= \frac{S_{xy}}{S_{xx}} = \frac{19.4}{10} = \boxed{1.94} \\
\hat{\beta}_0 &= \bar{y} - \hat{\beta}_1\bar{x} = 4.82 - 3.88 = \boxed{0.94}
\end{align*}

Fitted line: \(\hat{y} = 0.94 + 1.94x\)

\textbf{(b) Fit Statistics:}

\begin{small}
\begin{tabular}{ccccc}
\(x\) & \(y\) & \(\hat{y}\) & \(e\) & \(e^2\) \\
\hline
0 & 1.0 & 0.94 & 0.06 & 0.0036 \\
1 & 2.9 & 2.88 & 0.02 & 0.0004 \\
2 & 5.1 & 4.82 & 0.28 & 0.0784 \\
3 & 5.9 & 6.76 & -0.86 & 0.7396 \\
4 & 9.2 & 8.70 & 0.50 & 0.2500 \\
\hline
\end{tabular}
\end{small}

\(SSE = 1.072\), \(SSR = 37.636\)

\begin{align*}
R^2 &= \frac{SSR}{S_{yy}} = \frac{37.636}{38.708} = \boxed{0.9723} \\
s &= \sqrt{\frac{SSE}{n-2}} = \sqrt{\frac{1.072}{3}} = \boxed{0.598}
\end{align*}

\textbf{(c) Hypothesis Test:}

\(H_0: \beta_1 = 0\) vs \(H_a: \beta_1 \neq 0\)

\begin{align*}
SE(\hat{\beta}_1) &= \frac{s}{\sqrt{S_{xx}}} = \frac{0.598}{3.162} = 0.189 \\
t &= \frac{1.94}{0.189} = \boxed{10.265}
\end{align*}

\(df=3\), \(t_{0.025,3} = 3.182\)

Reject \(H_0\) (\(p \approx 0.002\))

95\% CI: \(1.94 \pm 3.182(0.189) = \boxed{(1.339, 2.541)}\)

\textbf{(d) Prediction Interval at \(x_0=5\):}

\begin{align*}
\hat{y}_0 &= 0.94 + 1.94(5) = \boxed{10.64} \\
SE_{pred} &= s\sqrt{1 + \frac{1}{n} + \frac{(x_0-\bar{x})^2}{S_{xx}}} \\
&= 0.598\sqrt{1 + 0.2 + 0.9} = \boxed{0.866}
\end{align*}

95\% PI: \(10.64 \pm 3.182(0.866) = \boxed{(7.88, 13.40)}\)
