\begin{frame}{Question 1: Tenure Status Among Visible Minorities (5 marks)}
    \textbf{Research Question:} Does tenure status differ for visible minorities?

    \vspace{0.3cm}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Methodology}
            \begin{enumerate}
                \item Calculate tenure percentages
                \item Create contingency table
                \item Perform $\chi^2$ test
                \item Calculate effect size (Cramér's V)
            \end{enumerate}

            \vspace{0.3cm}

            \textbf{Chi-Square Test Formula}
            \[
            \chi^2 = \sum \frac{(O_{ij} - E_{ij})^2}{E_{ij}}
            \]

            where:
            \begin{itemize}
                \item $O_{ij}$ = Observed frequency
                \item $E_{ij}$ = Expected frequency
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Key Results}

            \begin{table}[h]
                \centering
                \scriptsize
                \begin{tabular}{lcc}
                    \toprule
                    \textbf{Group} & \textbf{Tenured} & \textbf{Rate} \\
                    \midrule
                    Visible Minorities & Yes & XX.X\% \\
                    Non-Minorities & Yes & XX.X\% \\
                    \bottomrule
                \end{tabular}
            \end{table}

            \vspace{0.3cm}

            \textbf{Statistical Test}
            \begin{itemize}
                \item $H_0$: Independence
                \item $H_1$: Association exists
                \item $\alpha = 0.05$
                \item Decision: Based on p-value
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Question 2: Average Age by Tenure Status (4 marks)}
    \textbf{Research Question:} Does average age differ between tenured and non-tenured professors?

    \vspace{0.3cm}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Descriptive Statistics}

            \begin{table}[h]
                \centering
                \footnotesize
                \begin{tabular}{lcc}
                    \toprule
                    \textbf{Metric} & \textbf{Tenured} & \textbf{Non-Tenured} \\
                    \midrule
                    Mean Age & $\bar{x}_1$ & $\bar{x}_2$ \\
                    Std Dev & $s_1$ & $s_2$ \\
                    Sample Size & $n_1$ & $n_2$ \\
                    Min Age & - & - \\
                    Max Age & - & - \\
                    \bottomrule
                \end{tabular}
            \end{table}

            \vspace{0.3cm}

            \textbf{Two-Sample t-Test Formula}
            \[
            t = \frac{\bar{x}_1 - \bar{x}_2}{\sqrt{\frac{s_1^2}{n_1} + \frac{s_2^2}{n_2}}}
            \]
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Hypotheses}
            \begin{itemize}
                \item $H_0$: $\mu_1 = \mu_2$ (No difference)
                \item $H_1$: $\mu_1 \neq \mu_2$ (Significant difference)
            \end{itemize}

            \vspace{0.3cm}

            \textbf{Key Findings}
            \begin{itemize}
                \item Age difference: $\Delta = |\bar{x}_1 - \bar{x}_2|$ years
                \item Statistical significance at $\alpha = 0.05$
                \item Effect size (Cohen's d)
                \item Practical interpretation
            \end{itemize}

            \vspace{0.3cm}

            \textbf{Cohen's d Formula}
            \[
            d = \frac{\bar{x}_1 - \bar{x}_2}{s_{pooled}}
            \]
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Question 5: Median Evaluation Score (2 marks)}
    \textbf{Research Question:} What is the median evaluation score for tenured professors?

    \vspace{0.5cm}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Median Definition}

            \begin{itemize}
                \item \textbf{Definition:} Middle value in ordered data
                \item \textbf{Symbol:} $M$ or $\tilde{x}$
                \item \textbf{Properties:}
                    \begin{itemize}
                        \item Robust to outliers
                        \item 50th percentile
                        \item Divides data in half
                    \end{itemize}
            \end{itemize}

            \vspace{0.3cm}

            \textbf{Calculation Formula}

            For odd $n$:
            \[
            M = x_{\frac{n+1}{2}}
            \]

            For even $n$:
            \[
            M = \frac{x_{\frac{n}{2}} + x_{\frac{n}{2}+1}}{2}
            \]
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Results for Tenured Professors}
